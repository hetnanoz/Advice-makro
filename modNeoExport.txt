Private Const CLASS_NAME As String = "modNeoExport"


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       String
' Description:   UPDATE11: create a stable output folder with an incomplete-run marker.
'-------------------------------------------------------------------------------
Public Function CreateRunFolder(ByVal parentFolder As String, ByVal runStamp As String) As String
    Const METHOD_NAME As String = "CreateRunFolder"
    Dim fso As Object
    Dim candidate As String
    Dim suffix As Long
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FolderExists(parentFolder) Then
        Err.Raise TOOL_ERROR + 50, METHOD_NAME, "The output folder does not exist."
    End If
    candidate = fso.BuildPath(parentFolder, "NeoAdvices_" & runStamp)
    Do While fso.FolderExists(candidate)
        suffix = suffix + 1
        candidate = fso.BuildPath(parentFolder, _
            "NeoAdvices_" & runStamp & "_" & Format$(suffix, "000"))
    Loop
    If Len(candidate) > 125 Then
        Err.Raise TOOL_ERROR + 51, METHOD_NAME, "Choose a shorter output folder path."
    End If
    fso.CreateFolder candidate
    fso.CreateTextFile(fso.BuildPath(candidate, "IN_PROGRESS.txt"), False).Close
    CreateRunFolder = candidate

ExitPoint:
    Set fso = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       String
' Description:   UPDATE11: mark a completed batch without renaming its output folder.
'-------------------------------------------------------------------------------
Public Function FinalizeRunFolder(ByVal stagingFolder As String) As String
    Const METHOD_NAME As String = "FinalizeRunFolder"
    Dim fso As Object
    Dim pendingPath As String
    Dim completePath As String
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FolderExists(stagingFolder) Then
        Err.Raise TOOL_ERROR + 50, METHOD_NAME, "The output folder does not exist."
    End If
    pendingPath = fso.BuildPath(stagingFolder, "IN_PROGRESS.txt")
    completePath = fso.BuildPath(stagingFolder, "COMPLETE.txt")
    If Not fso.FileExists(pendingPath) Then
        Err.Raise TOOL_ERROR + 51, METHOD_NAME, "The incomplete-run marker is missing."
    End If
    ' Only the normal success path calls this function, after all output workbooks are closed.
    ' Publish completion last: a failed marker operation must still propagate to the caller.
    Name pendingPath As completePath
    FinalizeRunFolder = stagingFolder

ExitPoint:
    Set fso = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       Object
' Description:   Collect unique accountants and expected row counts from the enriched source.
'-------------------------------------------------------------------------------
Public Function AccountantCounts(ByVal wsNeo As Worksheet) As Object
    Const METHOD_NAME As String = "AccountantCounts"
    Dim result As Object
    Dim lastRow As Long
    Dim lastColumn As Long
    Dim values As Variant
    Dim rowIndex As Long
    Dim key As String
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set result = CreateObject("Scripting.Dictionary")
    result.CompareMode = vbTextCompare
    Call GetDataBounds(wsNeo, lastRow, lastColumn)
    values = wsNeo.Range(wsNeo.Cells(1, 1), wsNeo.Cells(lastRow, 1)).Value2
    For rowIndex = 2 To lastRow
        key = CleanText(values(rowIndex, 1))
        If Len(key) = 0 Then key = UNASSIGNED_NAME
        If result.Exists(key) Then
            result(key) = CLng(result(key)) + 1
        Else
            result.Add key, 1&
        End If
    Next rowIndex
    Set AccountantCounts = result

ExitPoint:
    Set result = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Delete non-owned table rows from bottom to top and validate retained ownership.
'-------------------------------------------------------------------------------
Public Sub TrimIndividualDashboard(ByVal wb As Workbook, ByVal accountantName As String, _
                                   ByVal expectedCount As Long, _
                                   Optional ByVal includeSources As Boolean = False)
    Const METHOD_NAME As String = "TrimIndividualDashboard"
    Dim ws As Worksheet
    Dim lo As ListObject
    Dim values As Variant
    Dim rowIndex As Long
    Dim fieldIndex As Long
    Dim actualCount As Long
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set ws = RequiredSheet(wb, SHEET_DASHBOARD)
    Set lo = ws.ListObjects(TABLE_DASHBOARD)
    Call RemoveAllSlicerCaches(wb)
    If lo.AutoFilter.FilterMode Then lo.AutoFilter.ShowAllData
    If lo.DataBodyRange Is Nothing Then
        Err.Raise TOOL_ERROR + 53, METHOD_NAME, "The copied Dashboard has no data rows."
    End If
    fieldIndex = lo.ListColumns(HDR_ACCOUNTANT).Index
    ' Rebuild the same rules once after deletion to avoid fragmented conditional formats.
    lo.DataBodyRange.FormatConditions.Delete
    values = lo.DataBodyRange.Value2
    For rowIndex = UBound(values, 1) To 1 Step -1
        If StrComp(CleanText(values(rowIndex, fieldIndex)), accountantName, vbTextCompare) <> 0 Then
            lo.ListRows(rowIndex).Delete
        End If
    Next rowIndex
    actualCount = lo.ListRows.Count
    If actualCount <> expectedCount Or actualCount = 0 Then
        Err.Raise TOOL_ERROR + 54, METHOD_NAME, "Individual Dashboard row-count validation failed."
    End If
    values = lo.DataBodyRange.Value2
    For rowIndex = 1 To UBound(values, 1)
        If StrComp(CleanText(values(rowIndex, fieldIndex)), accountantName, vbTextCompare) <> 0 Then
            Err.Raise TOOL_ERROR + 55, METHOD_NAME, "Individual Dashboard ownership validation failed."
        End If
    Next rowIndex
    Call ApplyDataFormatting(ws, DASH_HEADER_ROW, DASH_HEADER_ROW + actualCount, lo.ListColumns.Count, True)
    Call LoadAccountantLayout(wb, accountantName)
    Call CreateDashboardControls(wb, True)
    wb.Names.Add Name:=COPY_MARKER, RefersTo:="=TRUE", Visible:=False
    Call EnforceIndividualSheets(wb, includeSources)

ExitPoint:
    Set lo = Nothing
    Set ws = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Create an independent XLSM copy without touching the VBA project object model.
'-------------------------------------------------------------------------------
Private Sub SaveIndividualCopy(ByVal wbMaster As Workbook, ByVal filePath As String, _
                               ByVal accountantName As String, ByVal expectedCount As Long, _
                               ByVal allocationDate As Date, ByVal runStamp As String, _
                               ByVal includeSources As Boolean)
    Const METHOD_NAME As String = "SaveIndividualCopy"
    Dim wbIndividual As Workbook
    Dim fso As Object
    Dim copyCreated As Boolean
    Dim copyComplete As Boolean
    Dim cleanupFailed As Boolean
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set fso = CreateObject("Scripting.FileSystemObject")
    If fso.FileExists(filePath) Then
        Err.Raise TOOL_ERROR + 56, METHOD_NAME, "Output already exists; refusing to overwrite."
    End If
    wbMaster.SaveCopyAs filePath
    copyCreated = True
    Set wbIndividual = Application.Workbooks.Open(Filename:=filePath, UpdateLinks:=0, ReadOnly:=False, _
        IgnoreReadOnlyRecommended:=True, Notify:=False, AddToMru:=False)
    If wbIndividual.ReadOnly Then Err.Raise TOOL_ERROR + 57, METHOD_NAME, "Individual copy opened read-only."
    Call TrimIndividualDashboard(wbIndividual, accountantName, expectedCount, includeSources)
    Call WriteDashboardCaption(wbIndividual.Worksheets(SHEET_DASHBOARD), _
        allocationDate, runStamp, accountantName)
    With wbIndividual.Worksheets(SHEET_DASHBOARD).Cells(10, 1)
        If includeSources Then
            .Value2 = CStr(.Value2) & " | UPDATE08 | Source sheets: YES"
        Else
            .Value2 = CStr(.Value2) & " | UPDATE08 | Source sheets: NO"
        End If
    End With
    Call RebindDashboardButtons(wbIndividual, wbIndividual.Name)
    Call EnforceIndividualSheets(wbIndividual, includeSources)
    wbIndividual.Save
    wbIndividual.Close SaveChanges:=False
    Set wbIndividual = Nothing
    copyComplete = True

ExitPoint:
    Call CloseWorkbookSafe(wbIndividual, cleanupFailed)
    If copyCreated And Not copyComplete Then Call DeleteFileSafe(filePath)
    Set fso = Nothing
    If cleanupFailed And errNumber = 0 Then
        errNumber = TOOL_ERROR + 58
        errDescription = "Could not close the temporary individual workbook."
    End If
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Generate one timestamped workbook for each unique accountant, including UNASSIGNED.
'-------------------------------------------------------------------------------
Public Sub GenerateAccountantFiles(ByVal wbMaster As Workbook, ByVal folderPath As String, _
                                   ByVal runStamp As String, ByVal allocationDate As Date, _
                                   ByVal counts As Object)
    Const METHOD_NAME As String = "GenerateAccountantFiles"
    Dim accountant As Variant
    Dim filePath As String
    Dim completed As Long
    Dim includeSources As Boolean
    Dim wbTemplate As Workbook
    Dim workerExcel As Excel.Application
    Dim wbSource As Workbook
    Dim templatePath As String
    Dim templateFolder As String
    Dim fso As Object
    Dim stepDescription As String
    Dim templateCreated As Boolean
    Dim templateFolderCreated As Boolean
    Dim state As TExcelState
    Dim cleanupFailed As Boolean
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    includeSources = INCLUDE_SOURCES_IN_ACCOUNTANT_FILES
    Call CaptureExcelState(state)
    Application.DisplayAlerts = False
    Application.EnableEvents = False
    Application.AutomationSecurity = msoAutomationSecurityForceDisable
    Application.EnableCancelKey = xlErrorHandler
    Set wbSource = wbMaster
    If counts.Count > 0 And Not includeSources Then
        Application.Calculation = xlCalculationManual
        Set fso = CreateObject("Scripting.FileSystemObject")
        templateFolder = fso.BuildPath(fso.GetSpecialFolder(2).Path, "NeoAdvices_" & fso.GetTempName)
        fso.CreateFolder templateFolder
        templateFolderCreated = True
        templatePath = UniqueOutputPath(templateFolder, "NeoAdvices_TEMP_TEMPLATE_" & runStamp)
        stepDescription = "1/6 - saving template in system TEMP folder"
        Application.StatusBar = "NeoAdvices UPDATE08: " & stepDescription
        templateCreated = True
        wbMaster.SaveCopyAs templatePath
        stepDescription = "2/6 - opening template in isolated Excel instance"
        Application.StatusBar = "NeoAdvices UPDATE08: " & stepDescription
        Set workerExcel = CreateObject("Excel.Application")
        workerExcel.Visible = False
        workerExcel.DisplayAlerts = False
        workerExcel.ScreenUpdating = False
        workerExcel.EnableEvents = False
        workerExcel.AutomationSecurity = msoAutomationSecurityForceDisable
        Set wbTemplate = workerExcel.Workbooks.Open(Filename:=templatePath, UpdateLinks:=0, _
            ReadOnly:=False, IgnoreReadOnlyRecommended:=True, Notify:=False, AddToMru:=False)
        workerExcel.Calculation = xlCalculationManual
        If wbTemplate.ReadOnly Then
            Err.Raise TOOL_ERROR + 103, METHOD_NAME, "Export template opened read-only."
        End If
        If wbTemplate Is wbMaster Then
            Err.Raise TOOL_ERROR + 104, METHOD_NAME, "Refusing to change Master source sheets."
        End If
        ' Only the owned temporary copy loses source sheets. Dashboard and its event code remain.
        wbTemplate.Worksheets(SHEET_DASHBOARD).Visible = xlSheetVisible
        stepDescription = "3/6 - removing template slicer caches"
        Application.StatusBar = "NeoAdvices UPDATE08: " & stepDescription
        Call RemoveAllSlicerCaches(wbTemplate)
        stepDescription = "4/6 - deleting both source sheets (calculation suspended)"
        Application.StatusBar = "NeoAdvices UPDATE08: " & stepDescription
        wbTemplate.Worksheets(Array(SHEET_NEO, SHEET_REPARTITION)).Delete
        If wbTemplate.Sheets.Count <> 1 Then
            Err.Raise TOOL_ERROR + 105, METHOD_NAME, "Export template contains unexpected sheets."
        End If
        stepDescription = "5/6 - saving Dashboard-only template"
        Application.StatusBar = "NeoAdvices UPDATE08: " & stepDescription
        wbTemplate.Save
        Call CloseWorkbookSafe(wbTemplate, cleanupFailed)
        Call QuitExportExcelSafe(workerExcel, cleanupFailed)
        If cleanupFailed Then
            Err.Raise TOOL_ERROR + 113, METHOD_NAME, "Could not close the isolated template editor."
        End If
        stepDescription = "6/6 - opening prepared Dashboard-only template"
        Application.StatusBar = "NeoAdvices UPDATE08: " & stepDescription
        Set wbTemplate = Application.Workbooks.Open(Filename:=templatePath, UpdateLinks:=0, _
            ReadOnly:=False, IgnoreReadOnlyRecommended:=True, Notify:=False, AddToMru:=False)
        If wbTemplate.Sheets.Count <> 1 Then
            Err.Raise TOOL_ERROR + 114, METHOD_NAME, "Prepared template must contain only Dashboard."
        End If
        Set wbSource = wbTemplate
        Application.Calculation = state.Calculation
    End If
    For Each accountant In counts.Keys
        completed = completed + 1
        stepDescription = "accountant file " & CStr(completed) & " / " & CStr(counts.Count)
        Application.StatusBar = "NeoAdvices UPDATE08: " & stepDescription
        filePath = UniqueOutputPath(folderPath, _
            "NeoAdvices_" & CleanFileName(CStr(accountant)) & "_" & runStamp)
        Call SaveIndividualCopy(wbSource, filePath, CStr(accountant), _
            CLng(counts(accountant)), allocationDate, runStamp, includeSources)
    Next accountant

ExitPoint:
    Set wbSource = Nothing
    Call CloseWorkbookSafe(wbTemplate, cleanupFailed)
    Call QuitExportExcelSafe(workerExcel, cleanupFailed)
    If templateCreated Then
        If Not cleanupFailed Then Call DeleteExportTemplateSafe(templatePath, cleanupFailed)
    End If
    If templateFolderCreated Then
        If Not cleanupFailed Then Call RemoveExportFolderSafe(templateFolder, cleanupFailed)
    End If
    Set fso = Nothing
    Call RestoreExcelStateSafe(state, cleanupFailed)
    If cleanupFailed And errNumber = 0 Then
        errNumber = TOOL_ERROR + 106
        errDescription = "Could not clean up the export template or restore Excel state."
    End If
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription, stepDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    Exact owned temporary path and cumulative cleanup failure flag.
' Returns:       None.
' Description:   Do not publish a successful batch if its temporary template cannot be removed.
'-------------------------------------------------------------------------------
Private Sub DeleteExportTemplateSafe(ByVal filePath As String, ByRef hadFailure As Boolean)
    Dim fso As Object
    Dim cleanupNumber As Long
    Dim cleanupDescription As String

    On Error Resume Next
    VBA.Err.Clear
    Set fso = CreateObject("Scripting.FileSystemObject")
    If fso.FileExists(filePath) Then fso.DeleteFile filePath, False
    cleanupNumber = VBA.Err.Number
    cleanupDescription = VBA.Err.Description
    VBA.Err.Clear
    On Error GoTo 0
    Set fso = Nothing
    If cleanupNumber <> 0 Then
        hadFailure = True
        Call RecordError(CLASS_NAME, "DeleteExportTemplateSafe", cleanupNumber, cleanupDescription)
    End If
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    Marked individual copy and the policy captured in the generating Master.
' Returns:       None.
' Description:   Enforce the source-sheet policy on every individual file before saving (UPDATE06).
'-------------------------------------------------------------------------------
Private Sub EnforceIndividualSheets(ByVal wb As Workbook, ByVal includeSources As Boolean)
    Const METHOD_NAME As String = "EnforceIndividualSheets"
    Dim ws As Worksheet
    Dim sheetIndex As Long
    Dim state As TExcelState
    Dim cleanupFailed As Boolean
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If Not HasCopyMarker(wb) Then
        Err.Raise TOOL_ERROR + 110, METHOD_NAME, "Refusing to remove source sheets from an unmarked Master."
    End If
    Set ws = RequiredSheet(wb, SHEET_DASHBOARD)
    ws.Visible = xlSheetVisible
    If includeSources Then
        wb.Worksheets(SHEET_NEO).Visible = xlSheetHidden
        wb.Worksheets(SHEET_REPARTITION).Visible = xlSheetHidden
    Else
        Call CaptureExcelState(state)
        Application.DisplayAlerts = False
        Application.EnableEvents = False
        For sheetIndex = wb.Worksheets.Count To 1 Step -1
            Select Case wb.Worksheets(sheetIndex).Name
                Case SHEET_NEO, SHEET_REPARTITION
                    wb.Worksheets(sheetIndex).Delete
            End Select
        Next sheetIndex
        If wb.Sheets.Count <> 1 Then
            Err.Raise TOOL_ERROR + 111, METHOD_NAME, "UPDATE06: individual file must contain only Dashboard."
        End If
    End If

ExitPoint:
    Call RestoreExcelStateSafe(state, cleanupFailed)
    Set ws = Nothing
    If cleanupFailed And errNumber = 0 Then
        errNumber = TOOL_ERROR + 112
        errDescription = "Could not restore Excel state after applying the source-sheet policy."
    End If
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    Exact private TEMP folder created by this export, plus cleanup status.
' Returns:       None.
' Description:   Remove only the empty owned folder; never recursively delete its contents.
'-------------------------------------------------------------------------------
Private Sub RemoveExportFolderSafe(ByVal folderPath As String, ByRef hadFailure As Boolean)
    Dim cleanupNumber As Long
    Dim cleanupDescription As String

    On Error Resume Next
    VBA.Err.Clear
    RmDir folderPath
    cleanupNumber = VBA.Err.Number
    cleanupDescription = VBA.Err.Description
    VBA.Err.Clear
    On Error GoTo 0
    If cleanupNumber <> 0 Then
        hadFailure = True
        Call RecordError(CLASS_NAME, "RemoveExportFolderSafe", cleanupNumber, cleanupDescription)
    End If
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    Only the Excel instance created by this export, plus cumulative cleanup status.
' Returns:       None.
' Description:   Shut down the owned template editor without touching the user's Excel instance.
'-------------------------------------------------------------------------------
Private Sub QuitExportExcelSafe(ByRef workerExcel As Excel.Application, ByRef hadFailure As Boolean)
    Dim cleanupNumber As Long
    Dim cleanupDescription As String

    On Error Resume Next
    VBA.Err.Clear
    If Not workerExcel Is Nothing Then workerExcel.Quit
    cleanupNumber = VBA.Err.Number
    cleanupDescription = VBA.Err.Description
    VBA.Err.Clear
    On Error GoTo 0
    If cleanupNumber <> 0 Then
        hadFailure = True
        Call RecordError(CLASS_NAME, "QuitExportExcelSafe", cleanupNumber, cleanupDescription)
    Else
        Set workerExcel = Nothing
    End If
End Sub
