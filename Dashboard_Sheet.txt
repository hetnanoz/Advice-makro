Option Explicit

Private Const CLASS_NAME As String = "Dashboard_Sheet"
Private pLastSelection As Range

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See typed signature.
' Returns:       See typed signature.
' Description:   Worksheet_SelectionChange - UPDATE06.
'-------------------------------------------------------------------------------
Private Sub Worksheet_SelectionChange(ByVal Target As Range)
    Const METHOD_NAME As String = "Worksheet_SelectionChange"
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set pLastSelection = Target

ExitPoint:
    If errNumber <> 0 Then Call DisplayErrors
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka

' Parameters:    See typed signature.
' Returns:       See typed signature.
' Description:   Worksheet_Deactivate - UPDATE06.
'-------------------------------------------------------------------------------
Private Sub Worksheet_Deactivate()
    Const METHOD_NAME As String = "Worksheet_Deactivate"
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set pLastSelection = Nothing

ExitPoint:
    If errNumber <> 0 Then Call DisplayErrors
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)

End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See typed signature.
' Returns:       See typed signature.
' Description:   Worksheet_BeforeDoubleClick - UPDATE06.
'-------------------------------------------------------------------------------
Private Sub Worksheet_BeforeDoubleClick(ByVal Target As Range, Cancel As Boolean)
    Const METHOD_NAME As String = "Worksheet_BeforeDoubleClick"
    Dim lo As ListObject
    Dim candidate As ListObject
    Dim hit As Range
    Dim mode As Long
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Call ResetErrors
    If Target.CountLarge = 1 Then
        For Each candidate In Me.ListObjects
            If candidate.Name = TABLE_DASHBOARD Then Set lo = candidate
        Next candidate
        If Not lo Is Nothing Then
            If Not lo.DataBodyRange Is Nothing Then
                Set hit = Application.Intersect(Target, lo.DataBodyRange.Columns(1).Resize(, 2))
                If Not hit Is Nothing Then
                    Cancel = True
                    If Target.Column = lo.ListColumns("Done").Range.Column Then
                        If CleanText(Target.Value2) <> ChrW(&H2713) Then mode = 1
                    Else
                        If CleanText(Target.Value2) <> "?" Then mode = 2
                    End If
                    Call ApplyNeoStatus(Me, Target, mode)
                End If
            End If
        End If
    End If

ExitPoint:
    Set hit = Nothing
    Set candidate = Nothing
    Set lo = Nothing
    If errNumber <> 0 Then Call DisplayErrors
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See typed signature.
' Returns:       See typed signature.
' Description:   Worksheet_BeforeRightClick - UPDATE06.
'-------------------------------------------------------------------------------
Private Sub Worksheet_BeforeRightClick(ByVal Target As Range, Cancel As Boolean)
    Const METHOD_NAME As String = "Worksheet_BeforeRightClick"
    Dim lo As ListObject
    Dim candidate As ListObject
    Dim hit As Range
    Dim requested As Range
    Dim overlap As Range
    Dim mode As Long
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Call ResetErrors
    For Each candidate In Me.ListObjects
        If candidate.Name = TABLE_DASHBOARD Then Set lo = candidate
    Next candidate
    If Not lo Is Nothing Then
        If Not lo.DataBodyRange Is Nothing Then
            Set hit = Application.Intersect(Target, lo.DataBodyRange)
            If Not hit Is Nothing Then
                Set requested = Target
                If Not pLastSelection Is Nothing Then
                    Set overlap = Application.Intersect(Target, pLastSelection)
                    If Not overlap Is Nothing Then Set requested = pLastSelection
                End If
                Cancel = True
                mode = ChooseNeoStatus(Me.Parent)
                If mode >= 0 Then Call ApplyNeoStatus(Me, requested, mode)
            End If
        End If
    End If

ExitPoint:
    Set overlap = Nothing
    Set requested = Nothing
    Set hit = Nothing
    Set candidate = Nothing
    Set lo = Nothing
    If errNumber <> 0 Then Call DisplayErrors
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub
