Option Explicit

Private Const CLASS_NAME As String = "modNeoMain"


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Validate a saved, writable dedicated XLSM with its three persistent sheets.
'-------------------------------------------------------------------------------
Private Sub ValidateMaster(ByVal wb As Workbook)
    Const METHOD_NAME As String = "ValidateMaster"
    Dim ws As Worksheet
    Dim required As Variant
    Dim name As Variant
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If wb Is Nothing Then Err.Raise TOOL_ERROR + 70, METHOD_NAME, "No active Master workbook."
    If wb.FileFormat <> xlOpenXMLWorkbookMacroEnabled Or Len(wb.Path) = 0 Then
        Err.Raise TOOL_ERROR + 71, METHOD_NAME, "Save the Master as an XLSM file before running this tool."
    End If
    If wb.ReadOnly Then Err.Raise TOOL_ERROR + 72, METHOD_NAME, "The Master is read-only."
    If wb.ProtectStructure Then Err.Raise TOOL_ERROR + 73, METHOD_NAME, "Workbook structure is protected."
    If wb.Sheets.Count <> 3 Or wb.Worksheets.Count <> 3 Then
        Err.Raise TOOL_ERROR + 74, METHOD_NAME, "This tool requires exactly the three documented worksheets."
    End If
    If HasCopyMarker(wb) Then
        Err.Raise TOOL_ERROR + 75, METHOD_NAME, "Run the generator from the Master, not an accountant copy."
    End If
    required = Array(SHEET_DASHBOARD, SHEET_NEO, SHEET_REPARTITION)
    For Each name In required
        Set ws = RequiredSheet(wb, CStr(name))
        If ws.ProtectContents Or ws.ProtectDrawingObjects Then
            Err.Raise TOOL_ERROR + 76, METHOD_NAME, "A required worksheet is protected."
        End If
    Next name
    If Val(Application.Version) < 15 Then
        Err.Raise TOOL_ERROR + 77, METHOD_NAME, "Table slicers require Excel 2013 or newer for Windows."
    End If

ExitPoint:
    Set ws = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Validate selected file paths without rejecting open input workbooks.
'-------------------------------------------------------------------------------
Private Sub ValidateInputPaths(ByVal neoPath As String, ByVal repartitionPath As String)
    Const METHOD_NAME As String = "ValidateInputPaths"
    Dim fso As Object
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set fso = CreateObject("Scripting.FileSystemObject")
    If StrComp(neoPath, repartitionPath, vbTextCompare) = 0 Then
        Err.Raise TOOL_ERROR + 78, METHOD_NAME, "Select two different input files."
    End If
    If LCase$(fso.GetExtensionName(neoPath)) <> "xlsx" Or _
       LCase$(fso.GetExtensionName(repartitionPath)) <> "xlsx" Then
        Err.Raise TOOL_ERROR + 79, METHOD_NAME, "Both inputs must be XLSX files."
    End If
    If Not fso.FileExists(neoPath) Or Not fso.FileExists(repartitionPath) Then
        Err.Raise TOOL_ERROR + 80, METHOD_NAME, "An input file no longer exists."
    End If

ExitPoint:
    Set fso = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Validate, enrich, rebuild the Master and generate timestamped individual XLSM files.
'-------------------------------------------------------------------------------
Public Sub RunNeoAdvices()
    Const METHOD_NAME As String = "RunNeoAdvices"
    Dim wbMaster As Workbook
    Dim wbNeo As Workbook
    Dim wbRepartition As Workbook
    Dim neoOpenedByTool As Boolean
    Dim repartitionOpenedByTool As Boolean
    Dim wsNeoInput As Worksheet
    Dim wsRepInput As Worksheet
    Dim wsNeo As Worksheet
    Dim wsRep As Worksheet
    Dim wsDashboard As Worksheet
    Dim state As TExcelState
    Dim neoPath As String
    Dim repPath As String
    Dim outputFolder As String
    Dim stagingFolder As String
    Dim finalFolder As String
    Dim backupPath As String
    Dim runStamp As String
    Dim neoLastRow As Long
    Dim neoLastColumn As Long
    Dim repLastRow As Long
    Dim repLastColumn As Long
    Dim allocationColumn As Long
    Dim allocationDate As Date
    Dim sourceNeo As Variant
    Dim sourceRep As Variant
    Dim enriched As Variant
    Dim allocations As Object
    Dim counts As Object
    Dim selected As Boolean
    Dim proceed As Boolean
    Dim cleanupFailed As Boolean
    Dim completed As Boolean
    Dim unassignedCount As Long
    Dim progressAnswer As VbMsgBoxResult
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set wbMaster = Application.ActiveWorkbook
    Call ResetErrors
    Call ValidateMaster(wbMaster)
    Call CaptureExcelState(state)
    neoPath = PickInputFile("Wybierz plik 1: NeoAdvices Export (.xlsx)")
    If Len(neoPath) > 0 Then
        repPath = PickInputFile("Wybierz plik 2: WARSAW FA9 Repartition (.xlsx)")
        If Len(repPath) > 0 Then outputFolder = PickOutputFolder()
    End If
    If Len(neoPath) > 0 And Len(repPath) > 0 And Len(outputFolder) > 0 Then
        Call ValidateInputPaths(neoPath, repPath)
        Call SuspendExcel
        Application.StatusBar = "NeoAdvices: validating inputs..."
        Set wbNeo = AcquireInputWorkbook(neoPath, neoOpenedByTool)
        Set wbRepartition = AcquireInputWorkbook(repPath, repartitionOpenedByTool)
        If wbNeo.Date1904 <> wbMaster.Date1904 Or wbRepartition.Date1904 <> wbMaster.Date1904 Then
            Err.Raise TOOL_ERROR + 81, METHOD_NAME, "Inputs and Master must use the same Excel date system."
        End If
        Set wsNeoInput = ResolveInputSheet(wbNeo, True)
        Set wsRepInput = ResolveInputSheet(wbRepartition, False)
        Call GetDataBounds(wsNeoInput, neoLastRow, neoLastColumn)
        Call GetDataBounds(wsRepInput, repLastRow, repLastColumn)
        If neoLastRow < 2 Or repLastRow <= REP_HEADER_ROW Then
            Err.Raise TOOL_ERROR + 82, METHOD_NAME, "An input has headers but no data records."
        End If
        If neoLastRow + DASH_HEADER_ROW - 1 > wsNeoInput.Rows.Count Or _
           neoLastColumn + ADDED_COLUMNS + 2 > wsNeoInput.Columns.Count Then
            Err.Raise TOOL_ERROR + 83, METHOD_NAME, "The enriched output would exceed Excel worksheet limits."
        End If
        Call ValidateNeoHeaders(wsNeoInput, neoLastColumn)
        Call ValidateRepartition(wsRepInput)
        selected = ChooseAllocationColumn(wsRepInput, allocationColumn, allocationDate)
        If selected Then
            sourceNeo = wsNeoInput.Cells(1, 1).Resize(neoLastRow, neoLastColumn).Value2
            sourceRep = wsRepInput.Cells(1, 1).Resize(repLastRow, repLastColumn).Value2
            Set allocations = BuildAllocationDictionary(sourceRep, allocationColumn)
            enriched = BuildEnrichedData(wsNeoInput, sourceNeo, allocations)
            proceed = True
            If HasExistingProgress(wbMaster) Then
                progressAnswer = MsgBox("Obecny Dashboard zawiera Done/Doubt." & vbCrLf & _
                    "Nowy import wyzeruje te oznaczenia. Kontynuowac?" & vbCrLf & _
                    "Przed zmiana zostanie zapisana kopia Mastera.", _
                    vbYesNo + vbExclamation + vbDefaultButton2, "Reset Dashboardu")
                proceed = (progressAnswer = vbYes)
            End If
            If proceed Then
                runStamp = Format$(Now, "ddmmyyyyhhnnss")
                stagingFolder = CreateRunFolder(outputFolder, runStamp)
                backupPath = UniqueOutputPath(stagingFolder, "Master_BEFORE_RUN_" & runStamp)
                wbMaster.SaveCopyAs backupPath
                Application.StatusBar = "NeoAdvices: rebuilding Master..."
                Set wsNeo = RequiredSheet(wbMaster, SHEET_NEO)
                Set wsRep = RequiredSheet(wbMaster, SHEET_REPARTITION)
                Set wsDashboard = RequiredSheet(wbMaster, SHEET_DASHBOARD)
                Call RemoveAllSlicerCaches(wbMaster)
                Call ClearWorksheetView(wsDashboard)
                Call ClearWorksheetView(wsNeo)
                Call ClearWorksheetView(wsRep)
                wsDashboard.Visible = xlSheetVisible
                wsNeo.Visible = xlSheetVisible
                wsRep.Visible = xlSheetVisible
                Call CopyInputSnapshot(wsRepInput, wsRep, repLastRow, repLastColumn)
                Call CopyInputSnapshot(wsNeoInput, wsNeo, neoLastRow, neoLastColumn)
                Set wsNeoInput = Nothing
                Set wsRepInput = Nothing
                Call ReleaseInputWorkbookSafe(wbNeo, neoOpenedByTool, cleanupFailed)
                Call ReleaseInputWorkbookSafe(wbRepartition, repartitionOpenedByTool, cleanupFailed)
                If cleanupFailed Then
                    Err.Raise TOOL_ERROR + 94, METHOD_NAME, "Could not release a tool-opened input."
                End If
                wsNeo.Columns("A:G").Insert Shift:=xlToRight
                wsNeo.Columns("A:G").NumberFormat = "@"
                wsNeo.Cells(1, 1).Resize(UBound(enriched, 1), UBound(enriched, 2)).Value2 = enriched
                Call FormatHeader(wsNeo.Cells(1, 1).Resize(1, UBound(enriched, 2)))
                Call ApplyDataFormatting(wsNeo, 1, UBound(enriched, 1), UBound(enriched, 2), False)
                Call FitDataColumns(wsNeo, 1, UBound(enriched, 1), UBound(enriched, 2), False)
                Call BuildDashboard(wbMaster, enriched, allocationDate, runStamp)
                Set counts = AccountantCounts(wsNeo)
                If counts.Exists(UNASSIGNED_NAME) Then unassignedCount = CLng(counts(UNASSIGNED_NAME))
                ' Save workbooks with the user's calculation mode, not the temporary manual mode.
                Application.Calculation = state.Calculation
                wbMaster.Save
                Call SaveTimestampedMaster(wbMaster, stagingFolder, runStamp)
                Call GenerateAccountantFiles(wbMaster, stagingFolder, runStamp, allocationDate, counts)
                finalFolder = FinalizeRunFolder(stagingFolder)
                stagingFolder = vbNullString
                completed = True
            End If
        End If
    End If

ExitPoint:
    Set wsNeoInput = Nothing
    Set wsRepInput = Nothing
    Call ReleaseInputWorkbookSafe(wbNeo, neoOpenedByTool, cleanupFailed)
    Call ReleaseInputWorkbookSafe(wbRepartition, repartitionOpenedByTool, cleanupFailed)
    Call RestoreExcelStateSafe(state, cleanupFailed)
    If cleanupFailed And errNumber = 0 Then
        errNumber = TOOL_ERROR + 84
        errDescription = "Cleanup could not fully restore Excel state."
    End If
    If errNumber <> 0 And Len(stagingFolder) > 0 Then
        Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, _
            "An IN_PROGRESS output folder remains. The BEFORE_RUN snapshot preserves the previous Master.")
    End If
    If completed And errNumber = 0 Then
        MsgBox "Gotowe." & vbCrLf & "Instrukcje: " & CStr(neoLastRow - 1) & vbCrLf & _
            "Pliki Accountantow: " & CStr(counts.Count) & vbCrLf & _
            "UNASSIGNED: " & CStr(unassignedCount) & vbCrLf & _
            "Data alokacji: " & Format$(allocationDate, "dd/mm/yyyy") & vbCrLf & _
            "Folder: " & finalFolder, vbInformation, "NeoAdvices"
    End If
    Set counts = Nothing
    Set allocations = Nothing
    Set wsNeoInput = Nothing
    Set wsRepInput = Nothing
    Set wsNeo = Nothing
    Set wsRep = Nothing
    Set wsDashboard = Nothing
    Set wbMaster = Nothing
    If errNumber <> 0 Then
        Call DisplayErrors
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub
