'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Remove stale caches before replacing this dedicated three-sheet tool view.
'-------------------------------------------------------------------------------
Public Sub RemoveAllSlicerCaches(ByVal wb As Workbook)
    Const METHOD_NAME As String = "RemoveAllSlicerCaches"
    Dim cacheIndex As Long
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    For cacheIndex = wb.SlicerCaches.Count To 1 Step -1
        wb.SlicerCaches(cacheIndex).Delete
    Next cacheIndex

ExitPoint:
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Build the complete interactive table, leaving rows 1-10 for controls.
'-------------------------------------------------------------------------------
Public Sub BuildDashboard(ByVal wb As Workbook, ByRef enriched As Variant, _
                          ByVal allocationDate As Date, ByVal runStamp As String)
    Const METHOD_NAME As String = "BuildDashboard"
    Dim ws As Worksheet
    Dim result() As Variant
    Dim rowCount As Long
    Dim columnCount As Long
    Dim rowIndex As Long
    Dim columnIndex As Long
    Dim sourceIndex As Long
    Dim columnOrder As Variant
    Dim lo As ListObject
    Dim tableRange As Range
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set ws = RequiredSheet(wb, SHEET_DASHBOARD)
    rowCount = UBound(enriched, 1)
    columnCount = UBound(enriched, 2) + 2
    columnOrder = DefaultDashboardColumnOrder(columnCount)
    ReDim result(1 To rowCount, 1 To columnCount)
    result(1, 1) = "Done"
    result(1, 2) = "Doubt"
    For rowIndex = 1 To rowCount
        For columnIndex = 3 To columnCount
            sourceIndex = CLng(columnOrder(columnIndex)) - 2
            result(rowIndex, columnIndex) = enriched(rowIndex, sourceIndex)
        Next columnIndex
    Next rowIndex
    Set tableRange = ws.Cells(DASH_HEADER_ROW, 1).Resize(rowCount, UBound(result, 2))
    tableRange.NumberFormat = "@"
    tableRange.Value2 = result
    Set lo = ws.ListObjects.Add(SourceType:=xlSrcRange, Source:=tableRange, XlListObjectHasHeaders:=xlYes)
    lo.Name = TABLE_DASHBOARD
    ' Empty style is Excel's native no-style setting; TableStyleNone is not portable.
    lo.TableStyle = vbNullString
    lo.ShowTableStyleRowStripes = False
    lo.ShowTableStyleColumnStripes = False
    lo.ShowTotals = False
    ' Copy source formats using the same mapping as values; identifiers stay text.
    For columnIndex = 3 To columnCount
        sourceIndex = CLng(columnOrder(columnIndex)) - 2
        wb.Worksheets(SHEET_NEO).Cells(1, sourceIndex).Resize(rowCount, 1).Copy
        ws.Cells(DASH_HEADER_ROW, columnIndex).PasteSpecial Paste:=xlPasteFormats
    Next columnIndex
    Application.CutCopyMode = False
    Call SortDashboardRows(lo)
    Call FormatHeader(lo.HeaderRowRange)
    Call ApplyDataFormatting(ws, DASH_HEADER_ROW, DASH_HEADER_ROW + rowCount - 1, lo.ListColumns.Count, True)
    Call FitDataColumns(ws, DASH_HEADER_ROW, DASH_HEADER_ROW + rowCount - 1, lo.ListColumns.Count, True)
    ws.Rows("1:8").RowHeight = 24
    ws.Rows("9:10").RowHeight = 24
    Call CreateDashboardControls(wb, False)
    Call WriteDashboardCaption(ws, allocationDate, runStamp, "MASTER")

ExitPoint:
    Set tableRange = Nothing
    Set lo = Nothing
    Set ws = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Build native slicers and macro buttons without relying on active objects (UPDATE05).
'-------------------------------------------------------------------------------
Public Sub CreateDashboardControls(ByVal wb As Workbook, ByVal individual As Boolean)
    Const METHOD_NAME As String = "CreateDashboardControls"
    Dim ws As Worksheet
    Dim lo As ListObject
    Dim fields As Variant
    Dim captions As Variant
    Dim macroNames As Variant
    Dim fieldIndex As Long
    Dim buttonIndex As Long
    Dim itemIndex As Long
    Dim button As Shape
    Dim leftPosition As Double
    Dim topPosition As Double
    Dim buttonWidth As Double
    Dim ownerMacro As String
    Dim lastButton As Long
    Dim stepDescription As String
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    stepDescription = "Resolve Dashboard table"
    Set ws = RequiredSheet(wb, SHEET_DASHBOARD)
    Set lo = ws.ListObjects(TABLE_DASHBOARD)
    ws.Rows("1:8").RowHeight = 24
    ws.Rows("9:10").RowHeight = 24
    stepDescription = "Remove old controls"
    Call RemoveAllSlicerCaches(wb)
    For itemIndex = ws.Shapes.Count To 1 Step -1
        If Left$(ws.Shapes(itemIndex).Name, 7) = "btnNeo_" Then ws.Shapes(itemIndex).Delete
    Next itemIndex
    If individual Then
        fields = Array("KAG", "NAV Group", "NAV calculation", HDR_FUND)
    Else
        fields = Array("Accountant", "KAG", "NAV Group", "NAV calculation", HDR_FUND)
    End If
    For fieldIndex = LBound(fields) To UBound(fields)
        stepDescription = "Native slicer; field=" & CStr(fields(fieldIndex))
        Call CreateNativeTableSlicer(wb, ws, lo, CStr(fields(fieldIndex)), 6 + fieldIndex * 190)
    Next fieldIndex
    captions = Array("ALL", "Cash", "SecCash", "Security", "SecCash + Security")
    macroNames = Array("FilterAll", "FilterCash", "FilterSecCash", "FilterSecurity", "FilterSecCashSecurity")
    lastButton = 4
    leftPosition = 6
    topPosition = ws.Rows(9).Top + 1
    For buttonIndex = 0 To lastButton
        stepDescription = "Create button " & CStr(captions(buttonIndex))
        buttonWidth = 104
        If buttonIndex = 4 Then buttonWidth = 170
        If buttonIndex = 5 Then buttonWidth = 132
        Set button = ws.Shapes.AddShape(msoShapeRoundedRectangle, leftPosition, topPosition, buttonWidth, 21)
        button.Name = "btnNeo_" & CStr(buttonIndex + 1)
        stepDescription = "Format button " & CStr(captions(buttonIndex))
        With button.TextFrame
            .Characters.Text = CStr(captions(buttonIndex))
            .Characters.Font.Size = 10
            .Characters.Font.Bold = True
            .Characters.Font.Color = RGB(255, 255, 255)
            .HorizontalAlignment = xlHAlignCenter
            .VerticalAlignment = xlVAlignCenter
        End With
        button.Fill.ForeColor.RGB = RGB(31, 78, 121)
        button.Line.Visible = msoFalse
        button.Placement = xlFreeFloating
        stepDescription = "Bind macro " & CStr(macroNames(buttonIndex))
        ownerMacro = "'" & Replace(wb.Name, "'", "''") & "'!" & CStr(macroNames(buttonIndex))
        button.OnAction = ownerMacro
        leftPosition = leftPosition + buttonWidth + 6
    Next buttonIndex
    stepDescription = "Build progress counters"
    Call BuildProgressCounters(ws)

ExitPoint:
    Set button = Nothing
    Set lo = Nothing
    Set ws = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        stepDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Change only the Type Group filter; preserve selections in other slicers.
'-------------------------------------------------------------------------------
Public Sub ApplyTypeFilter(ByVal wb As Workbook, ByVal mode As Long)
    Const METHOD_NAME As String = "ApplyTypeFilter"
    Dim ws As Worksheet
    Dim lo As ListObject
    Dim fieldIndex As Long
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set ws = RequiredSheet(wb, SHEET_DASHBOARD)
    Set lo = ws.ListObjects(TABLE_DASHBOARD)
    fieldIndex = lo.ListColumns(HDR_GROUP).Index
    Select Case mode
        Case 0
            If lo.AutoFilter.Filters(fieldIndex).On Then lo.Range.AutoFilter Field:=fieldIndex
        Case 1
            lo.Range.AutoFilter Field:=fieldIndex, Criteria1:="Cash"
        Case 2
            lo.Range.AutoFilter Field:=fieldIndex, Criteria1:="SecCash"
        Case 3
            lo.Range.AutoFilter Field:=fieldIndex, Criteria1:="Security"
        Case 4
            lo.Range.AutoFilter Field:=fieldIndex, Criteria1:=Array("SecCash", "Security"), _
                Operator:=xlFilterValues
        Case Else
            Err.Raise TOOL_ERROR + 40, METHOD_NAME, "Unknown Type Group filter mode."
    End Select

ExitPoint:
    Set lo = Nothing
    Set ws = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       Boolean
' Description:   Detect Done or Doubt before a destructive refresh of the current view.
'-------------------------------------------------------------------------------
Public Function HasExistingProgress(ByVal wb As Workbook) As Boolean
    Const METHOD_NAME As String = "HasExistingProgress"
    Dim ws As Worksheet
    Dim lo As ListObject
    Dim values As Variant
    Dim rowIndex As Long
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set ws = RequiredSheet(wb, SHEET_DASHBOARD)
    For Each lo In ws.ListObjects
        If lo.Name = TABLE_DASHBOARD Then
            If Not lo.DataBodyRange Is Nothing Then
                values = lo.DataBodyRange.Columns(1).Resize(, 2).Value2
                For rowIndex = 1 To UBound(values, 1)
                    If CleanText(values(rowIndex, 1)) = ChrW(&H2713) Or _
                       CleanText(values(rowIndex, 2)) = "?" Then
                        HasExistingProgress = True
                        Exit For
                    End If
                Next rowIndex
            End If
            Exit For
        End If
    Next lo

ExitPoint:
    Set lo = Nothing
    Set ws = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    Explicit workbook/table/field and the factory method to probe.
' Returns:       A cache, or Nothing plus the preserved COM error.
' Description:   Isolated compatibility probe; no forced names in the namespace.
'-------------------------------------------------------------------------------
Private Function TryTableSlicerCache(ByVal wb As Workbook, ByVal lo As ListObject, _
                                     ByVal fieldName As String, ByVal legacyMethod As Boolean, _
                                     ByRef failureNumber As Long, _
                                     ByRef failureDescription As String) As SlicerCache
    failureNumber = 0
    failureDescription = vbNullString
    On Error Resume Next
    VBA.Err.Clear
    If legacyMethod Then
        Set TryTableSlicerCache = wb.SlicerCaches.Add(lo, fieldName)
    Else
        Set TryTableSlicerCache = wb.SlicerCaches.Add2(lo, fieldName)
    End If
    failureNumber = VBA.Err.Number
    failureDescription = VBA.Err.Description
    VBA.Err.Clear
    On Error GoTo 0
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    A newly created cache, explicit destination and argument form.
' Returns:       A slicer, or Nothing plus the preserved COM error.
' Description:   Probe only the factory call; allow Excel to choose a unique name.
'-------------------------------------------------------------------------------
Private Function TryTableSlicer(ByVal cache As SlicerCache, ByVal ws As Worksheet, _
                                ByVal useSheetName As Boolean, ByRef failureNumber As Long, _
                                ByRef failureDescription As String) As Slicer
    failureNumber = 0
    failureDescription = vbNullString
    On Error Resume Next
    VBA.Err.Clear
    If useSheetName Then
        Set TryTableSlicer = cache.Slicers.Add(ws.Name)
    Else
        Set TryTableSlicer = cache.Slicers.Add(ws)
    End If
    failureNumber = VBA.Err.Number
    failureDescription = VBA.Err.Description
    VBA.Err.Clear
    On Error GoTo 0
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    Cache created by the current failed slicer operation only.
' Returns:       None.
' Description:   Isolated cleanup; preserve the original creation error.
'-------------------------------------------------------------------------------
Private Sub DeleteNewSlicerCacheSafe(ByRef cache As SlicerCache)
    Dim cleanupNumber As Long
    Dim cleanupDescription As String

    On Error Resume Next
    VBA.Err.Clear
    If Not cache Is Nothing Then cache.Delete
    cleanupNumber = VBA.Err.Number
    cleanupDescription = VBA.Err.Description
    VBA.Err.Clear
    If cleanupNumber <> 0 Then
        Call RecordError(CLASS_NAME, "DeleteNewSlicerCacheSafe", cleanupNumber, cleanupDescription)
    End If
    Set cache = Nothing
    On Error GoTo 0
End Sub


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Create one native slicer with compatible factory calls and stage-level diagnostics.
'-------------------------------------------------------------------------------
Private Sub CreateNativeTableSlicer(ByVal wb As Workbook, ByVal ws As Worksheet, _
                                    ByVal lo As ListObject, ByVal fieldName As String, _
                                    ByVal leftPosition As Double)
    Const METHOD_NAME As String = "CreateNativeTableSlicer"
    Dim cache As SlicerCache
    Dim slicer As Slicer
    Dim stepDescription As String
    Dim firstFailure As String
    Dim failureNumber As Long
    Dim failureDescription As String
    Dim completed As Boolean
    Dim fieldIndex As Long
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    stepDescription = "Validate field " & fieldName
    fieldIndex = lo.ListColumns(fieldName).Index
    fieldName = lo.ListColumns(fieldIndex).Name
    If Not ws.Parent Is wb Then
        Err.Raise TOOL_ERROR + 95, METHOD_NAME, "Slicer destination belongs to another workbook."
    End If
    If Not lo.Parent Is ws Then
        Err.Raise TOOL_ERROR + 96, METHOD_NAME, "Slicer source table belongs to another worksheet."
    End If
    stepDescription = "Add2 cache; field=" & fieldName
    Set cache = TryTableSlicerCache(wb, lo, fieldName, False, failureNumber, failureDescription)
    If cache Is Nothing Then
        If failureNumber = 5 Or failureNumber = 1004 Or failureNumber = 438 Then
            firstFailure = "Add2 [" & CStr(failureNumber) & "]: " & failureDescription
            stepDescription = "Add cache; field=" & fieldName
            Set cache = TryTableSlicerCache(wb, lo, fieldName, True, failureNumber, failureDescription)
        End If
    End If
    If cache Is Nothing Then
        If failureNumber = 0 Then failureNumber = TOOL_ERROR + 97
        If Len(failureDescription) = 0 Then failureDescription = "Excel returned no slicer cache."
        Err.Raise failureNumber, METHOD_NAME, failureDescription
    End If
    firstFailure = vbNullString
    stepDescription = "Add slicer using worksheet object; field=" & fieldName
    Set slicer = TryTableSlicer(cache, ws, False, failureNumber, failureDescription)
    If slicer Is Nothing Then
        If failureNumber = 5 Or failureNumber = 1004 Or failureNumber = 438 Then
            firstFailure = "Worksheet argument [" & CStr(failureNumber) & "]: " & failureDescription
            stepDescription = "Add slicer using sheet name; field=" & fieldName
            Set slicer = TryTableSlicer(cache, ws, True, failureNumber, failureDescription)
        End If
    End If
    If slicer Is Nothing Then
        If failureNumber = 0 Then failureNumber = TOOL_ERROR + 99
        If Len(failureDescription) = 0 Then failureDescription = "Excel returned no slicer control."
        Err.Raise failureNumber, METHOD_NAME, failureDescription
    End If
    stepDescription = "Validate completed slicer source; field=" & fieldName
    Call ValidateTableSlicerSource(cache, lo)
    stepDescription = "Set caption; field=" & fieldName
    slicer.Caption = fieldName
    stepDescription = "Set layout; field=" & fieldName
    slicer.Top = ws.Rows(4).Top + 3
    slicer.Left = leftPosition
    slicer.Width = 184
    slicer.Height = 114
    slicer.NumberOfColumns = 1
    stepDescription = "Set placement; field=" & fieldName
    slicer.Shape.Placement = xlFreeFloating
    completed = True

ExitPoint:
    If Not completed Then Call DeleteNewSlicerCacheSafe(cache)
    Set slicer = Nothing
    Set cache = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        stepDescription & "; " & firstFailure)
    GoTo ExitPoint
End Sub


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    See the typed signature below.
' Returns:       None
' Description:   Validate the table identity after the native slicer has been created (UPDATE05).
'-------------------------------------------------------------------------------
Private Sub ValidateTableSlicerSource(ByVal cache As SlicerCache, ByVal expectedTable As ListObject)
    Const METHOD_NAME As String = "ValidateTableSlicerSource"
    Dim actualTable As ListObject
    Dim actualSheet As Worksheet
    Dim expectedSheet As Worksheet
    Dim actualBook As Workbook
    Dim expectedBook As Workbook
    Dim matches As Boolean
    Dim sourceContext As String
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    sourceContext = "Expected table=" & expectedTable.Name
    If Not cache.List Then
        Err.Raise TOOL_ERROR + 98, METHOD_NAME, "Excel created a non-table slicer cache."
    End If
    Set actualTable = cache.ListObject
    If actualTable Is Nothing Then
        Err.Raise TOOL_ERROR + 98, METHOD_NAME, "Excel returned no source table for the completed slicer."
    End If
    Set actualSheet = actualTable.Parent
    Set expectedSheet = expectedTable.Parent
    Set actualBook = actualSheet.Parent
    Set expectedBook = expectedSheet.Parent
    sourceContext = sourceContext & "; actual table=" & actualTable.Name & _
        "; actual sheet=" & actualSheet.Name
    ' Compare stable Excel identities instead of relying on COM object reference equality.
    matches = (StrComp(actualBook.FullName, expectedBook.FullName, vbTextCompare) = 0)
    matches = matches And (StrComp(actualSheet.Name, expectedSheet.Name, vbTextCompare) = 0)
    matches = matches And (StrComp(actualTable.Name, expectedTable.Name, vbTextCompare) = 0)
    matches = matches And (actualTable.Range.Address = expectedTable.Range.Address)
    If Not matches Then
        Err.Raise TOOL_ERROR + 98, METHOD_NAME, "The completed slicer is linked to a different table."
    End If

ExitPoint:
    Set actualTable = Nothing
    Set actualSheet = Nothing
    Set expectedSheet = Nothing
    Set actualBook = Nothing
    Set expectedBook = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription, sourceContext)
    GoTo ExitPoint
End Sub

' Returns:       None.
' Description:   Repair/test existing Dashboard controls without re-importing or exporting data.

Public Sub RebuildNeoControls()

    Dim wb As Workbook
    Dim state As TExcelState
    Dim cleanupFailed As Boolean
    Dim completed As Boolean
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set wb = Application.ActiveWorkbook
    Call ResetErrors
    If wb Is Nothing Then Err.Raise TOOL_ERROR + 100, METHOD_NAME, "No active workbook."
    Call CaptureExcelState(state)
    Application.EnableEvents = False
    Application.ScreenUpdating = False
    Call CreateDashboardControls(wb, HasCopyMarker(wb))
    completed = True

ExitPoint:
    Call RestoreExcelStateSafe(state, cleanupFailed)
    Set wb = Nothing
    If cleanupFailed And errNumber = 0 Then errNumber = TOOL_ERROR + 101
    If completed And errNumber = 0 Then
        MsgBox "Kontrolki Dashboardu zostaly odtworzone. Dane i Done/Doubt pozostaly bez zmian.", _
            vbInformation, "NeoAdvices - UPDATE05"
    End If
    If errNumber <> 0 Then
        Call DisplayErrors
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-14
' Parameters:    Complete Dashboard table, including Done and Doubt.
' Returns:       None.
' Description:   Keep funds together and sort ISINs ascending within each fund.
'-------------------------------------------------------------------------------
Public Sub SortDashboardRows(ByVal lo As ListObject)
    Const METHOD_NAME As String = "SortDashboardRows"
    Dim fundColumn As ListColumn
    Dim isinColumn As ListColumn
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set fundColumn = lo.ListColumns(HDR_FUND)
    Set isinColumn = lo.ListColumns(HDR_ISIN)
    If lo.ListRows.Count > 1 Then
        With lo.Sort
            .SortFields.Clear
            .SortFields.Add Key:=fundColumn.Range, SortOn:=xlSortOnValues, _
                Order:=xlAscending, DataOption:=xlSortNormal
            .SortFields.Add Key:=isinColumn.Range, SortOn:=xlSortOnValues, _
                Order:=xlAscending, DataOption:=xlSortNormal
            .Header = xlYes
            .MatchCase = False
            .Orientation = xlTopToBottom
            .Apply
        End With
    End If

ExitPoint:
    Set fundColumn = Nothing
    Set isinColumn = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-15
' Parameters:    columnCount: width of the original Dashboard layout including Done/Doubt.
' Returns:       One-based mapping of destination columns to original Dashboard columns.
' Description:   UPDATE13: preferred columns first; all remaining columns retain source order.
'-------------------------------------------------------------------------------
Private Function DefaultDashboardColumnOrder(ByVal columnCount As Long) As Variant
    Const METHOD_NAME As String = "DefaultDashboardColumnOrder"
    Dim preferred As Variant
    Dim ordered() As Long
    Dim used() As Boolean
    Dim itemIndex As Long
    Dim sourceIndex As Long
    Dim destinationIndex As Long
    Dim errNumber As Long
    Dim errDescription As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    ' Original layout: A, B, J, K, L, O, P, Q, U, V, W, X, Y, Z, AA.
    preferred = Array(1, 2, 10, 11, 12, 15, 16, 17, 21, 22, 23, 24, 25, 26, 27)
    If columnCount < 27 Then
        Err.Raise TOOL_ERROR + 126, METHOD_NAME, "Dashboard requires at least 27 source columns."
    End If
    ReDim ordered(1 To columnCount)
    ReDim used(1 To columnCount)
    For itemIndex = LBound(preferred) To UBound(preferred)
        sourceIndex = CLng(preferred(itemIndex))
        destinationIndex = destinationIndex + 1
        ordered(destinationIndex) = sourceIndex
        used(sourceIndex) = True
    Next itemIndex
    For sourceIndex = 1 To columnCount
        If Not used(sourceIndex) Then
            destinationIndex = destinationIndex + 1
            ordered(destinationIndex) = sourceIndex
        End If
    Next sourceIndex
    DefaultDashboardColumnOrder = ordered

ExitPoint:
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call RecordError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function
